% Restore the numbering used by the standalone worksheets.  The lecture-note
% header uses letters for top-level enumerations, whereas worksheet exercises
% are numbered and their subparts are lettered.
\newcommand{\worksheetnumbering}{%
  \renewcommand{\theenumi}{\arabic{enumi}}%
  \renewcommand{\labelenumi}{\theenumi.}%
  \renewcommand{\theenumii}{\alph{enumii}}%
  \renewcommand{\labelenumii}{(\theenumii)}%
}

% Imported worksheets already contain a document wrapper and a title.  Docmute
% removes the wrapper; this setup suppresses the repeated title and makes each
% Spanish heading subordinate to the surrounding worksheet section.
\newcommand{\worksheetimportsetup}{%
  \let\maketitle\relax
  \let\section\subsection
  \let\bf\bfseries
  \worksheetnumbering
}
