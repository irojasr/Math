Be sure to read each question carefully. You must choose and answer
\textbf{exactly two} of the four problems. If you attempt more than two, only
the first two will be graded. Write your final answers in the boxes provided.
Only answers written in the boxes will be graded. You may use the surrounding
space for scratch work. Each problem is worth the same number of points.

\begin{enumerate}

% Original two-to-one reversal: a parabolic boundary and a line.
\item Let $f$ be continuous. Rewrite
\[
  \int_0^1\int_0^{\sqrt{x}}f(x,y)\,dy\,dx
  +\int_1^2\int_0^{2-x}f(x,y)\,dy\,dx
\]
as a single double integral in the order $dx\,dy$. Do not evaluate.

\vfill
\begin{flushright}
\framebox(400,60){}
\end{flushright}

% Adapted from Moodle .008, VIT-V2, question 11712500:
% same solid, with density 1+z and a setup-only mass question.
\item A solid lies in the first octant, below $z=4-y^2$, with $x+y\leq2$.
Its density is $\rho(x,y,z)=1+z$. Set up a triple integral for its mass
in the order $dz\,dy\,dx$. Do not evaluate.

\vfill
\begin{flushright}
\framebox(400,60){}
\end{flushright}

\newpage

% Shortlist candidate C: an original linear-bound reversal with a split at y=1.
\item Let $h$ be continuous. Rewrite
\[
  \int_0^1\int_x^{2-x}h(x,y)\,dy\,dx
\]
in the order $dx\,dy$. Do not evaluate.

\vfill
\begin{flushright}
\framebox(400,60){}
\end{flushright}

% Shortlist candidate B, adapted from Moodle .008, question 11522301:
% endpoint values supplied for a numerical answer; no order change needed.
\item Let $g$ be continuously differentiable, with $g(0)=1$ and $g(1)=4$.
Evaluate
\[
  \int_0^1\int_0^1\int_0^{\sqrt{2x}}2g'(y)z\,dz\,dx\,dy.
\]

\vfill
\begin{flushright}
\framebox(250,50){}
\end{flushright}

\end{enumerate}
