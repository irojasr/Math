Be sure to read each question carefully. You must choose and answer
\textbf{exactly two} of the four problems. If you attempt more than two, only
the first two will be graded. Write your final answers in the boxes provided.
Only answers written in the boxes will be graded. You may use the surrounding
space for scratch work. Each problem is worth the same number of points.

\begin{enumerate}

\item Consider the lines
\[
  L_1:\ \vec{r}_1(t)=(1,0,2)+t(1,2,-1),\qquad
  L_2:\ \vec{r}_2(s)=(0,1,4)+s(2,4,-2).
\]
Answer yes or no:
\begin{enumerate}[(a)]
  \item Are $L_1$ and $L_2$ parallel?
  \item Do $L_1$ and $L_2$ intersect? If so, where?
\end{enumerate}

\vfill
\begin{flushright}
(a)\ \framebox(180,50){}\qquad
(b)\ \framebox(180,50){}
\end{flushright}

\item Write down a counterclockwise parametrization
$\vec{r}(t)$ of the ellipse
\[
  {x^2}+4y^2=25,\qquad 0\leq t< 2\pi.
\]
With that in hand, compute the tangent vector $\vec{v}(\pi/2)$.

\vfill
\begin{flushright}
\framebox(250,50){}
\end{flushright}

\newpage

\item Find a parametric equation $\vec{r}(t)$ for the line of intersection
of the planes
\[
  x+3y+z=0
  \qquad\text{and}\qquad
  x+6y+2z=0.
\]
\textit{Hint: Take the cross product of the planes' normal vectors to obtain
a direction vector. To find a point on the intersection, set $y=0$ and solve
for $x$ and $z$. Then use $\vec{r}(t)=t\vec{v}+P$.}

\vfill
\begin{flushright}
\framebox(250,50){}
\end{flushright}

\item A fly follows the path
\[
  \vec{r}(t)=(t,t^2,2t),\qquad t\geq 0.
\]
It does not see a transparent sheet of sticky film lying in the plane
$x+y+z=10$. (a) At what time does the fly hit the film, and (b) at what point does it
stick?

\vfill
\begin{flushright}
(a)\ \framebox(180,50){}\qquad
(b)\ \framebox(180,50){}
\end{flushright}

\end{enumerate}
