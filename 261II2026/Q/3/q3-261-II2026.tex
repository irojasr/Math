% Adapted from Moodle, ``Derivadas Parciales 1.''
\item The state variables of an ideal gas satisfy
\[
  PV=nRT,
\]
where $n$ and $R$ are constants. Compute
\[
  \left(\frac{\partial V}{\partial T}\right)
  \left(\frac{\partial T}{\partial P}\right)
  \left(\frac{\partial P}{\partial V}\right).
\]

\vfill
\begin{flushright}
\framebox(250,50){}
\end{flushright}

% From the UCR partial-derivatives worksheet, Exercise 3.6.
\item Let $z=f(u,v)$, where $f$ is twice differentiable, and let
\[
  u=3x^2y,\qquad v=4xy^2.
\]
Compute $\dfrac{\partial^2 z}{\partial x^2}$ in terms of $x$, $y$, and the
partial derivatives $f_u$, $f_{uu}$, $f_{uv}$, and $f_{vv}$.

\vfill
\begin{flushright}
\framebox(400,60){}
\end{flushright}

% Numerical chain-rule problem inspired by the UCR and Moodle chain-rule sets.
\item Let $z=f(x,y)$, where
\[
  x=t^2+1,\qquad y=2t-1.
\]
Given that $\nabla f(2,1)=(3,-2)$, compute $\dfrac{dz}{dt}$ at $t=1$.

\vfill
\begin{flushright}
\framebox(250,50){}
\end{flushright}

% Adapted from Moodle, ``Dev Dir 1.''
\item The temperature in a room is
\[
  T(x,y,z)=9x^2-3y^2+6xyz.
\]
A fly is at $(2,2,2)$. Find the unit vector in the direction the fly should
move so that the temperature decreases most rapidly.

\vfill
\begin{flushright}
\framebox(250,50){}
\end{flushright}

\end{enumerate}
